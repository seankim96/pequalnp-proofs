let differences : Finset (Fin 13) := {0,1,3,9}
let reps : List (Nat × Nat) := [(7,904),(13,2848),(19,708),(21,2720),(25,2656),(37,2440),(41,2370),(133,808)]
let multipliers : List (Fin 13) := [1,3,9]
let flag (p w : Fin 13) : Fin 26 → Bool :=
  fun i => decide (i = Fin.castAdd 13 p ∨ i = Fin.natAdd 13 w)
let orbit (z : Nat × Nat) (a t : Fin 13) (dual : Bool) : Fin 26 → Bool :=
  let bit := fun (u v : Nat) (i : Fin 13) =>
    if dual then v.testBit (-(a*i+t)).val else u.testBit (a*i+t).val
  Fin.append (bit z.1 z.2) (bit z.2 z.1)
let covers {m : Nat} (pick : Fin m → Nat) (gs : List PEqualNP.CircuitLowerBounds.Gate) :=
  ∀ g ∈ gs, ∀ i j, g.op = PEqualNP.CircuitLowerBounds.GateOp.and → g.inputs = [i,j] → i ≠ j →
    ∃ k, pick k = i ∨ pick k = j
∀ C : PEqualNP.CircuitLowerBounds.Circuit, PEqualNP.MetaComplexity.Monotone2 C →
    (∀ p w : Fin 13, (∃ d ∈ differences, p = w+d) → PEqualNP.MetaComplexity.circuitFun 26 C (flag p w) = true) →
    (∀ z ∈ reps, ∀ a ∈ multipliers, ∀ t d, PEqualNP.MetaComplexity.circuitFun 26 C (orbit z a t d) = false) →
    ∀ (m : Nat) (pick : Fin m → Nat), covers pick C.gates →
      9 ≤ m ∧ (m = 9 → ∀ k, 26 ≤ pick k) ∧ ∃ rows : Fin 468 → (Fin 26 → Bool),
        (∀ j, ∃ z ∈ reps, ∃ a ∈ multipliers, ∃ t d, rows j = orbit z a t d) ∧
        Function.Injective (fun (j : Fin 468) (k : Fin m) =>
          (C.wires (List.ofFn (rows j))).getD (pick k) false)
