let Column := Nat → Bool
let disj (ws : List Column) (r : Nat) := ws.any (fun w => w r)
let both (a b : Column) (r : Nat) := a r && b r
let wire (s : List Column) (i : Nat) := s.getD i (fun _ => false)
let addOr (s : List Column) (g : Nat × Nat) := s ++ [fun r => wire s g.1 r || wire s g.2 r]
let good (p a b : Nat) (w : Column) := w p = true → w a = true ∨ w b = true
let separated (s : List Column) := ∃ n : Fin 257 → Fin 468, ∀ i j : Fin 257, i.val < j.val →
    ∃ p : Fin 52, ∀ w ∈ s, good p.val (52 + (n i).val) (52 + (n j).val) w
let Gate := Bool × Nat × Nat
let step (s : List Column) (g : Gate) : List Column :=
  if g.1 then s ++ [both (wire s g.2.1) (wire s g.2.2)] else addOr s g.2
∀ (base : List Column)
    (ors : List (Nat × Nat)) (left right : Nat) (tail : List Gate) (out : Nat),
    (∀ l r : List Column, (∀ w ∈ l, w ∈ base) → (∀ w ∈ r, w ∈ base) →
      separated (base ++ [both (disj l) (disj r)])) →
    let s := ors.foldl addOr base
    let final := tail.foldl step (s ++ [both (wire s left) (wire s right)])
    (∀ p : Fin 52, wire final out p.val = true) →
    (∀ n : Fin 468, wire final out (52 + n.val) = false) →
    9 ≤ tail.countP Prod.fst
