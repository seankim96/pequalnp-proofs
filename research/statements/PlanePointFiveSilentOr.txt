let Column := Nat → Bool
let Gate := Bool × Nat × Nat
let wire (s : List Column) (i : Nat) := s.getD i (fun _ => false)
let mix (op : Bool) (x y : Column) : Column := fun r => if op then x r && y r else x r || y r
let newWire (s : List Column) (g : Gate) := mix g.1 (wire s g.2.1) (wire s g.2.2)
let step (s : List Column) (g : Gate) := s ++ [newWire s g]
let flag (k : Fin 3) (i : Fin 19) : Nat :=
  ((#[#[8194, 16386, 32772, 65544, 65552, 65600, 69632, 131073, 131104, 131200, 262146, 524292, 1048832, 1049600, 2097664, 4194816, 8390656, 16777218, 33554436], #[8194, 16386, 32772, 69632, 131073, 131088, 131104, 131200, 262146, 524292, 1048584, 1048832, 1049600, 2097664, 4194816, 8388672, 8390656, 16777218, 33554436], #[8194, 16386, 32772, 65544, 65552, 65600, 69632, 131073, 131104, 131200, 262146, 524292, 1048832, 1049600, 2097664, 4194816, 8390656, 16777218, 33554436]] : Array (Array Nat))[k.val]!)[i.val]!
let negative (k : Fin 3) : Nat := (#[1247750, 9572870, 9634310] : Array Nat)[k.val]!
let initial : List Column :=
  let raw := List.ofFn (fun i : Fin 26 => fun r : Nat => r.testBit i.val)
  let base := ([(false,0,1), (false,3,9), (true,15,27), (true,17,26), (true,13,26), (true,13,27), (false,2,6), (true,1,24), (true,15,32), (true,25,32), (true,0,17), (true,0,23), (true,0,25), (false,17,19), (false,20,24), (false,39,40), (true,7,41)] : List Gate).foldl step raw
  base ++ [mix false (wire base 2) (wire base 5)]
∀ (k : Fin 3) (pre post : List Gate) (g : Gate) (o : Nat),
    let gs := pre ++ g :: post
    (∀ p : Fin 19, wire (gs.foldl step initial) o (flag k p) = true) →
    wire (gs.foldl step initial) o (negative k) = false →
    gs.countP (fun h => !h.1) ≤ 18 → g.1 = false →
    newWire (pre.foldl step initial) g (negative k) = false
