let Column := Nat → Bool
let Gate := Bool × Nat × Nat
let wire (s : List Column) (i : Nat) := s.getD i (fun _ => false)
let mix (op : Bool) (x y : Column) : Column := fun r => if op then x r && y r else x r || y r
let newWire (s : List Column) (g : Gate) := mix g.1 (wire s g.2.1) (wire s g.2.2)
let step (s : List Column) (g : Gate) := s ++ [newWire s g]
let flag (i : Fin 19) : Nat := (#[8194, 16386, 32772, 32776, 32800, 34816, 65600, 131073, 131088, 131200, 262146, 524352, 1048832, 1049600, 2097664, 4194816, 8388672, 16777218, 33558528] : Array Nat)[i.val]!
let negative : Nat := 1217090
let initial : List Column :=
  let raw := List.ofFn (fun i : Fin 26 => fun r : Nat => r.testBit i.val)
  ([(false,0,1), (false,3,9), (true,15,27), (true,17,26), (true,13,26), (true,13,27), (false,2,6), (true,1,24), (true,15,32), (true,25,32), (true,0,17), (true,0,23), (true,0,25), (false,17,19), (false,20,24)] : List Gate).foldl step raw
∀ (gs : List Gate) (o : Nat),
    (∀ p : Fin 19, wire (gs.foldl step initial) o (flag p) = true) →
    wire (gs.foldl step initial) o negative = false →
    18 ≤ gs.countP (fun g => !g.1)
