let flag (p w : Fin 13) : Fin 26 → Bool :=
  fun i => decide (i = Fin.castAdd 13 p ∨ i = Fin.natAdd 13 w)
let orbit (z : Nat × Nat) (a t : Fin 13) (dual : Bool) : Fin 26 → Bool :=
  let bit := fun (u v : Nat) (i : Fin 13) =>
    if dual then v.testBit (-(a*i+t)).val else u.testBit (a*i+t).val
  Fin.append (bit z.1 z.2) (bit z.2 z.1)
let andPair (g : PEqualNP.CircuitLowerBounds.Gate) : Option (Nat × Nat) :=
  let i := g.inputs.getD 0 0
  let j := g.inputs.getD 1 0
  let isAnd := PEqualNP.CircuitLowerBounds.GateOp.rec (motive := fun _ => Bool)
    true false false (fun _ => false) false g.op
  if isAnd = true ∧ g.inputs.length = 2 ∧ i ≠ j then some (i,j) else none
let operands (gs : List PEqualNP.CircuitLowerBounds.Gate) := gs.filterMap andPair
let select (gs : List PEqualNP.CircuitLowerBounds.Gate) (right : Fin (operands gs).length → Bool)
    (k : Fin (operands gs).length) : Nat :=
  if right k then ((operands gs).get k).2 else ((operands gs).get k).1
