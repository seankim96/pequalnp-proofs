let wire (x : Nat) (w : Fin 31) : Bool :=
  if w.val < 26 then x.testBit w.val
  else if w.val = 26 then x.testBit 0 && x.testBit 13
  else if w.val = 27 then x.testBit 0 || x.testBit 14
  else if w.val = 28 then x.testBit 1 && x.testBit 13
  else if w.val = 29 then x.testBit 1 && x.testBit 18
  else x.testBit 2 && x.testBit 15
let product (a : Fin 31) (rest : List (Fin 31)) (x : Nat) :=
  wire x a && rest.all (wire x)
let Term := Fin 31 × List (Fin 31)
let value (ts : List Term) (x : Nat) := ts.any (fun t => product t.1 t.2 x)
let inherited (ws : List (Fin 31)) (x : Nat) := ws.any (wire x)
∀ ts : List Term,
    ∃ flags : List (Fin 52), flags.length ≤ ts.length ∧
      ∀ (p : Fin 52) (a b : Nat), p ∉ flags →
        (∀ w : Fin 31, wire positive[p.val]! w = true → wire a w = true ∨ wire b w = true) →
        value ts positive[p.val]! = true → value ts a = true ∨ value ts b = true
