let C := Nat → Bool
let G := Bool × Nat × Nat
let wire (s : List C) (j : Nat) := s.getD j (fun _ => false)
let eval (s : List C) (g : G) : C := fun r =>
  if g.1 then wire s g.2.1 r && wire s g.2.2 r else wire s g.2.1 r || wire s g.2.2 r
let step (s : List C) (g : G) := s ++ [eval s g]
let better (a b : C) :=
  (∀ p : Fin 52, a p.val = true → b p.val = true) ∧
  (∀ j : Fin 468, b (52+j.val) = true → a (52+j.val) = true)
let redirect (cut alt j : Nat) := if j < cut then j else if j = cut then alt else j-1
let rename (cut alt : Nat) (g : G) : G := (g.1,redirect cut alt g.2.1,redirect cut alt g.2.2)
let legal (n : Nat) (gs : List G) := ∀ i : Fin gs.length,
  let g := gs[i.val]'i.isLt
  g.2.1 < n+i.val ∧ g.2.2 < n+i.val
let guard (k : Nat) (g : G) := g.1 && (g.2.1 == k || g.2.2 == k)
∀ (s : List C) (g : G) (tail : List G) (alt : Nat), alt < s.length →
  better (eval s g) (wire s alt) → legal (s.length+1) tail →
  let ys := tail.map (rename s.length alt)
  legal s.length ys ∧ ys.length + 1 = (g :: tail).length ∧
  (∀ o, better (wire (tail.foldl step (step s g)) o)
    (wire (ys.foldl step s) (redirect s.length alt o))) ∧
  ∀ k < s.length, tail.countP (guard k) ≤ ys.countP (guard k)
