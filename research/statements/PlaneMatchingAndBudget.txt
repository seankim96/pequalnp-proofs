let Column := Nat → Bool
let disj (ws : List Column) (r : Nat) := ws.any (fun w => w r)
let both (a b : Column) (r : Nat) := a r && b r
let wire (s : List Column) (i : Nat) := s.getD i (fun _ => false)
let addOr (s : List Column) (g : Nat × Nat) := s ++ [fun r => wire s g.1 r || wire s g.2 r]
let good (p a b : Nat) (w : Column) := w p = true → w a = true ∨ w b = true
let separated (s : List Column) := ∃ n : Fin 257 → Fin 468, ∀ i j : Fin 257, i.val < j.val →
    ∃ p : Fin 52, ∀ w ∈ s, good p.val (52 + (n i).val) (52 + (n j).val) w
let Gate := Bool × Nat × Nat
let step (s : List Column) (g : Gate) : List Column :=
  if g.1 then s ++ [both (wire s g.2.1) (wire s g.2.2)] else addOr s g.2
let clashes (s : Finset (Fin 468)) (mate : Fin 468 → Fin 468) :=
  s.filter fun i => mate i < i ∧ mate i ∈ s
let side (l r : Column) : Finset (Fin 468) :=
  Finset.univ.filter fun i => l (52+i.val) = true → r (52+i.val) = true
∀ (s : List Column) (l r : Column) (mate : Fin 468 → Fin 468)
    (tail : List Gate) (o : Nat),
    (∀ a b : Fin 468, a < b → mate b ≠ a →
      ∃ p : Fin 52, ∀ w ∈ s, good p.val (52+a.val) (52+b.val) w) →
    (∀ p a b : Nat, (∀ w ∈ s, good p a b w) → good p a b l ∧ good p a b r) →
    (257 + (clashes (side l r) mate).card ≤ (side l r).card ∨
     257 + (clashes (side r l) mate).card ≤ (side r l).card) →
    let final := tail.foldl step (s ++ [both l r])
    (∀ p : Fin 52, wire final o p.val = true) →
    (∀ n : Fin 468, wire final o (52+n.val) = false) →
    9 ≤ tail.countP Prod.fst
