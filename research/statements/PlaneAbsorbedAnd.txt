      let D : PEqualNP.CircuitLowerBounds.Circuit := ⟨pre ++ (⟨.or,[i]⟩ : PEqualNP.CircuitLowerBounds.Gate) :: post,out⟩

        (∀ r, N r → D.eval r = false)
